\probheader{1}{Lanzamiento horizontal desde una ventana}{azul}
\databox{$v_0$: 10 m/s (horizontal)  $|$  $t$: 5 s  $|$  $v_{0y}$: 0 m/s}
\begin{center}
  \includegraphics[width=14cm]{/home/andres/Documentos/ANTIGRAVITY-PROYECTOS/FISICA/img_apuntes/p1.png}
\end{center}
\paso{Concepto clave}
El movimiento se separa en \textbf{dos ejes independientes}: horizontal (MRU, sin aceleración)
y vertical (caída libre desde reposo, $v_{0y} = 0$).
\paso{a) Altura de la ventana}
En el eje vertical, la única aceleración es la gravedad:
\begin{align*}
  h &= \frac{1}{2} g t^2 = \frac{1}{2} \times 9.8 \times 5^2 \\
    &= \frac{1}{2} \times 9.8 \times 25 = 122.5\ \text{m}
\end{align*}
\resbox{Altura de la ventana}{$h = 122.5$ m}
\paso{b) Distancia horizontal}
En el eje horizontal no hay aceleración (MRU):
\begin{align*}
  x &= v_0 \cdot t = 10 \times 5 = 50\ \text{m}
\end{align*}
\resbox{Distancia horizontal}{$x = 50$ m}

\probheader{2}{Tiro parabólico — bala disparada a 35°}{coral}
\databox{$v_0$: 400 m/s  $|$  $\theta$: 35°}
\begin{center}
  \includegraphics[width=14cm]{/home/andres/Documentos/ANTIGRAVITY-PROYECTOS/FISICA/img_apuntes/p2.png}
\end{center}
\paso{Componentes de la velocidad inicial}
\begin{align*}
  v_{0x} &= v_0 \cos\theta = 400 \cos 35^\circ \approx 327.66\ \text{m/s} \\
  v_{0y} &= v_0 \sin\theta = 400 \sin 35^\circ \approx 229.43\ \text{m/s}
\end{align*}
\paso{a) Tiempo total en el aire}
En el punto más alto $v_y = 0$. El vuelo es simétrico:
\begin{align*}
  t_{\text{vuelo}} &= \frac{2 v_{0y}}{g} = \frac{2 \times 229.43}{9.8} \approx 46.82\ \text{s}
\end{align*}
\resbox{Tiempo en el aire}{$t \approx 46.82$ s}
\paso{b) Altura máxima}
\begin{align*}
  H &= \frac{v_{0y}^2}{2g} = \frac{229.43^2}{2 \times 9.8} = \frac{52637}{19.6} \\
    &\approx 2685.6\ \text{m} \approx 2.69\ \text{km}
\end{align*}
\resbox{Altura máxima}{$H \approx 2685.6$ m ($\approx 2.69$ km)}
\paso{c) Alcance horizontal}
\begin{align*}
  R &= \frac{v_0^2 \sin(2\theta)}{g} = \frac{400^2 \times \sin 70^\circ}{9.8} \\
    &= \frac{160000 \times 0.9397}{9.8} \approx 15342\ \text{m} \approx 15.34\ \text{km}
\end{align*}
\resbox{Alcance horizontal}{$R \approx 15342$ m (15.34 km)}
\vspace{6pt}\hrule height 0.5pt\vspace{10pt}
\probheader{3}{Ángulo de elevación para alcanzar 4000 m}{amber}
\databox{$v_0$: 350 m/s  $|$  $R$: 4000 m}
\begin{center}
  \includegraphics[width=14cm]{/home/andres/Documentos/ANTIGRAVITY-PROYECTOS/FISICA/img_apuntes/p3.png}
\end{center}
\paso{Despejar el ángulo de la fórmula de alcance}
Partimos de $R = v_0^2 \sin(2\theta) / g$ y despejamos:
\begin{align*}
  \sin(2\theta) &= \frac{R \cdot g}{v_0^2} = \frac{4000 \times 9.8}{350^2} \\
               &= \frac{39200}{122500} \approx 0.32
\end{align*}
\paso{Dos soluciones del arcoseno}
La función seno tiene dos valores en $[0^\circ, 180^\circ]$:
\begin{align*}
  2\theta_1 &= \arcsin(0.32) \approx 18.66^\circ \quad\to\quad \theta_1 \approx 9.33^\circ \\
  2\theta_2 &= 180^\circ - 18.66^\circ = 161.34^\circ \quad\to\quad \theta_2 \approx 80.67^\circ
\end{align*}
\resbox{Ángulos válidos}{$\theta_1 = 9.33^\circ$ (rasante)  $\cdot$  $\theta_2 = 80.67^\circ$ (parábola alta)}
\begin{tcolorbox}[colback=grisclr, colframe=gris, arc=4pt, boxrule=0.3pt,
  left=8pt, right=8pt, top=4pt, bottom=4pt, width=\textwidth]
  \small\textcolor{gris}{Ambos ángulos producen exactamente el mismo alcance.}
\end{tcolorbox}

\probheader{4}{Proyectil lanzado desde un avión a 800 km/h}{gris}
\databox{$v$: 800 km/h $\to$ 222.22 m/s  $|$  $h$: 500 m}
\begin{center}
  \includegraphics[width=14cm]{/home/andres/Documentos/ANTIGRAVITY-PROYECTOS/FISICA/img_apuntes/p4.png}
\end{center}
\paso{Conversión de unidades — paso obligatorio}
Siempre trabaja en SI. Convertir km/h a m/s:
\begin{align*}
  v &= 800 \times \frac{1000}{3600} = 222.22\ \text{m/s}
\end{align*}
\paso{a) Tiempo de caída}
El proyectil se lanza horizontalmente ($v_{0y} = 0$):
\begin{align*}
  h &= \frac{1}{2} g t^2 \\
  t &= \sqrt{\frac{2h}{g}} = \sqrt{\frac{2 \times 500}{9.8}} = \sqrt{102.04} \approx 10.10\ \text{s}
\end{align*}
\resbox{Tiempo de caída}{$t \approx 10.10$ s}
\paso{b) Distancia horizontal}
\begin{align*}
  x &= v \cdot t = 222.22 \times 10.10 \approx 2244\ \text{m} \approx 2.24\ \text{km}
\end{align*}
\resbox{Distancia horizontal}{$x \approx 2244$ m (2.24 km)}

\probheader{5}{Tiro parabólico — balón de fútbol a 40°}{teal}
\databox{$v_0$: 22 m/s  $|$  $\theta$: 40°}
\begin{center}
  \includegraphics[width=14cm]{/home/andres/Documentos/ANTIGRAVITY-PROYECTOS/FISICA/img_apuntes/p5.png}
\end{center}
\paso{Componentes de la velocidad inicial}
\begin{align*}
  v_{0x} &= 22 \cos 40^\circ \approx 16.85\ \text{m/s} \\
  v_{0y} &= 22 \sin 40^\circ \approx 14.14\ \text{m/s}
\end{align*}
\paso{a) Altura máxima}
\begin{align*}
  H &= \frac{v_{0y}^2}{2g} = \frac{14.14^2}{2 \times 9.8} = \frac{199.94}{19.6} \\
    &\approx 10.20\ \text{m}
\end{align*}
\resbox{Altura máxima}{$H \approx 10.20$ m}
\paso{b) Alcance horizontal}
\begin{align*}
  R &= \frac{v_0^2 \sin(2\theta)}{g} = \frac{22^2 \times \sin 80^\circ}{9.8} \\
    &= \frac{484 \times 0.9848}{9.8} \approx 48.63\ \text{m}
\end{align*}
\resbox{Alcance horizontal}{$R \approx 48.63$ m}
