\section{Indicadores de Éxito (SMART)}
\begin{itemize}
    \item \textbf{Capacitación:} 100\% de tasa de aprobación de los 15 estudiantes.
    \item \textbf{Impacto:} 300 personas capacitadas indirectamente en Chihuahua.
    \item \textbf{Vinculación:} 3 vínculos formales establecidos con instituciones taiwanesas.
    \item \textbf{Inserción:} 80\% de empleabilidad en sectores de alta tecnología tras 6 meses.
\end{itemize}
